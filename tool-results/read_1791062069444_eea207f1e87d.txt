    79→    79→/**
    80→    80→ * Word-boundary truncation: never cuts mid-word ("Acute Kidney Inju…" →
    81→    81→ * "Acute Kidney…"), prefers a space, falls back to a hyphen boundary
    82→    82→ * ("Renin-Angiotensin…" → "Renin-…"), and strips dangling separators from
    83→    83→ * names like "Heart Failure (HF)" → "Heart Failure…".
    84→    84→ */
    85→    85→function truncateLabel(s: string, max: number = LABEL_MAX_CHARS): string {
    86→    86→  if (s.length <= max) return s
    87→    87→  const head = s.slice(0, max)
    88→    88→  const space = head.lastIndexOf(' ')
    89→    89→  if (space > 3) return `${head.slice(0, space).replace(/[\s(—/:;,&]+$/, '')}…`
    90→    90→  const hyphen = head.lastIndexOf('-')
    91→    91→  if (hyphen > 2) return `${head.slice(0, hyphen + 1)}…`
    92→    92→  return `${head.trimEnd()}…`
    93→    93→}

   118→   118→export function MedicalMapView() {
   119→   119→  const conceptFocus = useAppStore(s => s.conceptFocus)
   120→   120→  const openConcept = useAppStore(s => s.openConcept)
   121→   121→  const reduceMotion = useReducedMotion()
   122→   122→
   123→   123→  // Arriving from a focused concept (e.g. opened from Home) → neighborhood scope
   124→   124→  const [scope, setScope] = useState(() => (conceptFocus ? `concept:${conceptFocus}` : 'all'))
   125→   125→  const [data, setData] = useState<GraphPayload | null>(null)
   126→   126→  const [loading, setLoading] = useState(true)
   127→   127→  const [error, setError] = useState<string | null>(null)
   128→   128→  const [subjects, setSubjects] = useState<SubjectSummary[]>([])
   129→   129→  const [hover, setHover] = useState<{ node: MapNode; left: number; top: number; containerW: number } | null>(null)
   130→   130→  const [dragging, setDragging] = useState(false)
   131→   131→  const [view, setView] = useState({ k: 1, x: 0, y: 0 })
   132→   132→  // Client-side status quick-filter — 'attention' = weak/unstable, 'unlearned' = new
   133→   133→  const [statusFilter, setStatusFilter] = useState<StatusFilter>('all')
   134→   134→
   135→   135→  const containerRef = useRef<HTMLDivElement>(null)
   136→   136→  const svgRef = useRef<SVGSVGElement>(null)
   137→   137→  const gRef = useRef<SVGGElement>(null)
   138→   138→  const dragRef = useRef<{ px: number; py: number; vx: number; vy: number } | null>(null)
   139→   139→  const movedRef = useRef(false)
   140→   140→  const lastOpenRef = useRef(0)
   141→   141→
   142→   142→  const load = useCallback(async (s: string) => {
   143→   143→    setLoading(true)
   144→   144→    setError(null)
   145→   145→    try {
   146→   146→      const d = await api.graph(s)
   147→   147→      setData(d)
   148→   148→    } catch (e) {
   149→   149→      setError(e instanceof Error ? e.message : 'Failed to load the medical map')
   150→   150→    } finally {
   151→   151→      setLoading(false)
   152→   152→    }
   153→   153→  }, [])
   154→   154→
   155→   155→  useEffect(() => { void load(scope) }, [scope, load])
   156→   156→
   157→   157→  // Reset viewport whenever the scope changes
   158→   158→  useEffect(() => {
   159→   159→    setView({ k: 1, x: 0, y: 0 })
   160→   160→    setHover(null)
   161→   161→  }, [scope])
   162→   162→
   163→   163→  useEffect(() => {
   164→   164→    let cancelled = false
   165→   165→    api.subjects()
   166→   166→      .then(d => { if (!cancelled) setSubjects(d.subjects) })
   167→   167→      .catch(() => { /* subject filter is optional — ignore fetch failure */ })
   168→   168→    return () => { cancelled = true }
   169→   169→  }, [])
   170→   170→
   171→   171→  // Zoom around the cursor — viewBox-unit math (letterbox-aware)
   172→   172→  useEffect(() => {
   173→   173→    const svg = svgRef.current
   174→   174→    if (!svg) return
   175→   175→    const onWheel = (e: WheelEvent) => {
   176→   176→      e.preventDefault()
   177→   177→      const rect = svg.getBoundingClientRect()
   178→   178→      const s = Math.min(rect.width / VIEW_W, rect.height / VIEW_H)
   179→   179→      const ox = (rect.width - VIEW_W * s) / 2
   180→   180→      const oy = (rect.height - VIEW_H * s) / 2
   181→   181→      const cx = (e.clientX - rect.left - ox) / s
   182→   182→      const cy = (e.clientY - rect.top - oy) / s
   183→   183→      setView(prev => {
   184→   184→        const k = clamp(prev.k * Math.exp(-e.deltaY * 0.0016), MIN_ZOOM, MAX_ZOOM)
   185→   185→        const ratio = k / prev.k
   186→   186→        return { k, x: cx - (cx - prev.x) * ratio, y: cy - (cy - prev.y) * ratio }
   187→   187→      })
   188→   188→    }
   189→   189→    svg.addEventListener('wheel', onWheel, { passive: false })
   190→   190→    return () => svg.removeEventListener('wheel', onWheel)
   191→   191→  }, [])
   192→   192→
   193→   193→  const onPointerDown = (e: ReactPointerEvent<SVGSVGElement>) => {
   194→   194→    if (e.pointerType === 'mouse' && e.button !== 0) return
   195→   195→    if (dragRef.current) return // already panning (multi-touch guard)
   196→   196→    dragRef.current = { px: e.clientX, py: e.clientY, vx: view.x, vy: view.y }
   197→   197→    movedRef.current = false
   198→   198→    setDragging(true)
   199→   199→    setHover(null)
   200→   200→  }
   201→   201→
   202→   202→  // Drag lifecycle as a window-listener effect: no pointer capture (which would
   203→   203→  // retarget `click` away from nodes), and works for mouse + touch alike.
   204→   204→  useEffect(() => {
   205→   205→    if (!dragging) return
   206→   206→    const start = dragRef.current
   207→   207→    if (!start) return
   208→   208→    const onMove = (ev: PointerEvent) => {
   209→   209→      const dx = ev.clientX - start.px
   210→   210→      const dy = ev.clientY - start.py
   211→   211→      if (Math.abs(dx) + Math.abs(dy) > 6) movedRef.current = true
   212→   212→      const svg = svgRef.current
   213→   213→      if (!svg) return
   214→   214→      const rect = svg.getBoundingClientRect()
   215→   215→      const s = Math.min(rect.width / VIEW_W, rect.height / VIEW_H)
   216→   216→      setView(prev => ({ k: prev.k, x: start.vx + dx / s, y: start.vy + dy / s }))
   217→   217→    }
   218→   218→    const onUp = () => {
   219→   219→      dragRef.current = null
   220→   220→      setDragging(false)
   221→   221→    }
   222→   222→    window.addEventListener('pointermove', onMove)
   223→   223→    window.addEventListener('pointerup', onUp)
   224→   224→    window.addEventListener('pointercancel', onUp)
   225→   225→    return () => {
   226→   226→      window.removeEventListener('pointermove', onMove)
   227→   227→      window.removeEventListener('pointerup', onUp)
   228→   228→      window.removeEventListener('pointercancel', onUp)
   229→   229→    }
   230→   230→  }, [dragging])
   231→   231→
   232→   232→  const onNodeClick = (id: string) => {
   233→   233→    if (movedRef.current) return
   234→   234→    const now = Date.now()
   235→   235→    if (now - lastOpenRef.current < 350) return // ignore double-click spam (would open+close)
   236→   236→    lastOpenRef.current = now
   237→   237→    openConcept(id)
   238→   238→  }
   239→   239→
   240→   240→  const showHover = useCallback((n: LaidNode) => {
   241→   241→    if (dragRef.current) return // suppress tooltips while panning
   242→   242→    const g = gRef.current
   243→   243→    const container = containerRef.current
   244→   244→    if (!g || !container) return
   245→   245→    const ctm = g.getScreenCTM()
   246→   246→    if (!ctm) return
   247→   247→    const pt = new DOMPoint(n.x, n.y).matrixTransform(ctm)
   248→   248→    const rect = container.getBoundingClientRect()
   249→   249→    setHover({ node: n.node, left: pt.x - rect.left, top: pt.y - rect.top, containerW: rect.width })
   250→   250→  }, [])
   251→   251→
   252→   252→  const layout = useMemo(() => {
   253→   253→    if (!data) return null
   254→   254→    return forceLayout(
   255→   255→      data.nodes.map(n => ({ id: n.id, r: radiusOf(n) })),
   256→   256→      data.edges.map(e => ({ from: e.from, to: e.to })),
   257→   257→      { width: VIEW_W, height: VIEW_H, seed: SEED },
   258→   258→    )
   259→   259→  }, [data])
   260→   260→
   261→   261→  // Layout positions are computed for the full payload (so toggling filters never
   262→   262→  // reflows the graph), then the status filter hides nodes + touching edges.
   263→   263→  const nodes: LaidNode[] = useMemo(() => {
   264→   264→    if (!data || !layout) return []
   265→   265→    const laid = data.nodes.flatMap(n => {
   266→   266→      const pos = layout.get(n.id)
   267→   267→      return pos ? [{ node: n, x: pos.x, y: pos.y }] : []
   268→   268→    })
   269→   269→    const visible = laid.filter(n =>
   270→   270→      statusFilter === 'all'
   271→   271→        ? true
   272→   272→        : statusFilter === 'attention'
   273→   273→          ? n.node.status === 'weak' || n.node.status === 'unstable'
   274→   274→          : n.node.status === 'new',
   275→   275→    )
   276→   276→    // Label collision softening (deterministic, no mutation): among visible nodes
   277→   277→    // sharing a y-band, alternate the label above/below by parity of preceding
   278→   278→    // collisions. Edge-of-canvas nodes flip toward the safe side.
   279→   279→    return visible.map((n, i) => {
   280→   280→      const stacked = visible.reduce(
   281→   281→        (cnt, p, j) => (j < i && Math.abs(p.y - n.y) < LABEL_COLLIDE_Y ? cnt + 1 : cnt),
   282→   282→        0,
   283→   283→      )
   284→   284→      const nearTop = n.y < 34
   285→   285→      const nearBottom = n.y > VIEW_H - 26
   286→   286→      return {
   287→   287→        ...n,
   288→   288→        labelAbove: nearTop ? false : nearBottom ? true : stacked % 2 === 0,
   289→   289→      }
   290→   290→    })
   291→   291→  }, [data, layout, statusFilter])
   292→   292→
   293→   293→  const edges = useMemo(() => {
   294→   294→    if (!data || !layout) return []
   295→   295→    const visibleIds = new Set(nodes.map(n => n.node.id))
   296→   296→    return data.edges
   297→   297→      .map((e, i) => {
   298→   298→        const a = layout.get(e.from)
   299→   299→        const b = layout.get(e.to)
   300→   300→        if (!a || !b) return null
   301→   301→        return { key: `${e.from}-${e.to}-${e.type}-${i}`, ...e, x1: a.x, y1: a.y, x2: b.x, y2: b.y }
   302→   302→      })
   303→   303→      .filter((e): e is NonNullable<typeof e> => e !== null && visibleIds.has(e.from) && visibleIds.has(e.to))
   304→   304→  }, [data, layout, nodes])
   305→   305→
   306→   306→  const centerId = scope.startsWith('concept:') ? scope.slice(8) : null
   307→   307→  const centerName = centerId ? data?.nodes.find(n => n.id === centerId)?.name ?? 'Focused concept' : null
   308→   308→  const subjectSel = scope.startsWith('subject:') ? scope.slice(8) : '__all__'
   309→   309→
   310→   310→  // Zoom-aware label decluttering: zoomed out, only major nodes keep labels.
   311→   311→  const showAllLabels = view.k >= LABEL_MIN_ZOOM
   312→   312→  const isMajorNode = (n: MapNode) =>
   313→   313→    radiusOf(n) >= MAJOR_RADIUS || n.status === 'weak' || n.status === 'unstable' || n.id === centerId
   314→   314→
   315→   315→  return (
   316→   316→    <div className="space-y-4">
   317→   317→      {/* ── Header ── */}
   318→   318→      <header className="flex items-start gap-3">
   319→   319→        <div className="flex size-10 shrink-0 items-center justify-center rounded-xl border border-line bg-surface-2 text-primary md:size-11">
   320→   320→          <Waypoints className="size-5" />
   321→   321→        </div>
   322→   322→        <div>
   323→   323→          <h1 className="text-xl font-semibold tracking-tight md:text-2xl">MEDICAL MAP</h1>
   324→   324→          <p className="text-xs text-ink-soft md:text-sm">Travel the connections between concepts, diseases and drugs.</p>
   325→   325→        </div>
   326→   326→      </header>
   327→   327→
   328→   328→      {/* ── Scope bar ── */}
   329→   329→      <div className="flex flex-wrap items-center gap-2">
   330→   330→        <ScopeChip active={scope === 'all'} onClick={() => setScope('all')}>All</ScopeChip>
   331→   331→        {SYSTEMS.map(sys => {
   332→   332→          const Icon = SYSTEM_ICONS[sys.icon]
   333→   333→          const active = scope === `system:${sys.id}`
   334→   334→          return (
   335→   335→            <ScopeChip key={sys.id} active={active} onClick={() => setScope(active ? 'all' : `system:${sys.id}`)}>
   336→   336→              {Icon && <Icon className="size-3.5" />}
   337→   337→              {sys.label}
   338→   338→            </ScopeChip>
   339→   339→          )
   340→   340→        })}
   341→   341→        {/* status quick-filter chips (client-side only) */}
   342→   342→        <span aria-hidden className="mx-1 hidden h-6 w-px shrink-0 bg-line md:block" />
   343→   343→        <ScopeChip active={statusFilter === 'all'} onClick={() => setStatusFilter('all')}>ALL</ScopeChip>
   344→   344→        <ScopeChip active={statusFilter === 'attention'} onClick={() => setStatusFilter('attention')}>
   345→   345→          <span aria-hidden className="size-2 rounded-full" style={{ backgroundColor: 'var(--sev-warn)' }} />
   346→   346→          NEEDS WORK
   347→   347→        </ScopeChip>
   348→   348→        <ScopeChip active={statusFilter === 'unlearned'} onClick={() => setStatusFilter('unlearned')}>
   349→   349→          <span aria-hidden className="size-2 rounded-full" style={{ backgroundColor: 'var(--muted-foreground)' }} />
   350→   350→          NOT YET LEARNED
   351→   351→        </ScopeChip>
   352→   352→        <div className="ml-auto">
   353→   353→          <Select
   354→   354→            value={subjectSel}
   355→   355→            onValueChange={v => setScope(v === '__all__' ? 'all' : `subject:${v}`)}
   356→   356→          >
   357→   357→            <SelectTrigger className="h-9 w-[170px] text-xs md:w-[210px]" aria-label="Filter map by subject">
   358→   358→              <SelectValue placeholder="Subject view" />
   359→   359→            </SelectTrigger>
   360→   360→            <SelectContent>
   361→   361→              <SelectItem value="__all__">All subjects</SelectItem>
   362→   362→              {subjects.map(s => (
   363→   363→                <SelectItem key={s.id} value={s.id}>{s.name}</SelectItem>
   364→   364→              ))}
   365→   365→            </SelectContent>
   366→   366→          </Select>
   367→   367→        </div>
   368→   368→      </div>
   369→   369→
   370→   370→      {/* ── Legend + node count ── */}
   371→   371→      <div className="flex flex-wrap items-center gap-x-4 gap-y-2 text-[11px] text-ink-soft">
   372→   372→        <div className="flex flex-wrap items-center gap-x-3 gap-y-1">
   373→   373→          {Object.entries(KIND_META).map(([kind, meta]) => (
   374→   374→            <span key={kind} className="inline-flex items-center gap-1">
   375→   375→              <span className="size-2 rounded-full" style={{ backgroundColor: meta.color }} />
   376→   376→              {meta.label}
   377→   377→            </span>
   378→   378→          ))}
   379→   379→        </div>
   380→   380→        <span className="inline-flex flex-wrap items-center gap-x-3 gap-y-1">
   381→   381→          {STATUS_ORDER.map(s => (
   382→   382→            <span key={s} className="inline-flex items-center gap-1">
   383→   383→              <span className="size-2.5 rounded-full border-2 bg-transparent" style={{ borderColor: STATUS_COLORS[s] }} />
   384→   384→              {STATUS_LABELS[s]}
   385→   385→            </span>
   386→   386→          ))}
   387→   387→        </span>
   388→   388→        {data && !loading && (
   389→   389→          <span className="ml-auto font-medium text-foreground/75">
   390→   390→            {statusFilter === 'all'
   391→   391→              ? `${nodes.length} nodes · ${edges.length} links`
   392→   392→              : `${nodes.length} shown · ${data.nodes.length} mapped`}
   393→   393→          </span>
   394→   394→        )}
   395→   395→      </div>
   396→   396→
   397→   397→      {/* ── Graph canvas ── */}
   398→   398→      <div
   399→   399→        ref={containerRef}
   400→   400→        className="relative h-[520px] overflow-hidden rounded-3xl border border-line bg-surface-2 med-grid md:h-[640px]"
   401→   401→        onDoubleClick={() => setView({ k: 1, x: 0, y: 0 })}
   402→   402→      >
