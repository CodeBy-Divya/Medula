   102→function Segmented({
   103→  options,
   104→  value,
   105→  onChange,
   106→  ariaLabel,
   107→}: {
   108→  options: { value: number; label: string }[]
   109→  value: number | null
   110→  onChange: (v: number) => void
   111→  ariaLabel: string
   112→}) {
   113→  return (
   114→    <div role="radiogroup" aria-label={ariaLabel} className="grid grid-cols-2 gap-2 sm:grid-cols-3">
   115→      {options.map((o) => (
   116→        <button
   117→          key={o.value}
   118→          type="button"
   119→          role="radio"
   120→          aria-checked={value === o.value}
   121→          onClick={() => onChange(o.value)}
   122→          className={cn(
   123→            'min-h-11 rounded-xl border px-3 text-sm font-medium transition-all',
   124→            value === o.value
   125→              ? 'border-primary/60 bg-primary/10 text-foreground shadow-[0_0_20px_-8px_rgba(34,211,238,0.55)]'
   126→              : 'border-line bg-surface-2 text-ink-soft hover:border-primary/40 hover:text-foreground'
   127→          )}
   128→        >
   129→          {o.label}
   130→        </button>
   131→      ))}
   132→    </div>
   133→  )
   134→}

   195→export function OnboardingWizard() {
   196→  const setView = useAppStore((s) => s.setView)
   197→  const setProfile = useAppStore((s) => s.setProfile)
   198→  const { toast } = useToast()
   199→  const reduce = usePrefersReducedMotion()
   200→
   201→  const [step, setStep] = useState(0)
   202→  const [dir, setDir] = useState(1)
   203→  const [saving, setSaving] = useState(false)
   204→  const [skipping, setSkipping] = useState(false)
   205→  const [error, setError] = useState<string | null>(null)
   206→
   207→  const [name, setName] = useState('Future Dr.')
   208→  const [year, setYear] = useState<number | null>(null)
   209→  const [semester, setSemester] = useState(1)
   210→  const [collegeName, setCollegeName] = useState('')
   211→  const [collegeType, setCollegeType] = useState<CollegeType>('Government')
   212→  const [gradYear, setGradYear] = useState(() => new Date().getFullYear() + 3)
   213→  const [pastScore, setPastScore] = useState('')
   214→  const [prepStage, setPrepStage] = useState<Profile['prepStage'] | null>(null)
   215→  const [dailyHours, setDailyHours] = useState(4)
   216→  const [weekdayHours, setWeekdayHours] = useState(5)
   217→  const [weekendHours, setWeekendHours] = useState(8)
   218→  const [learningStyles, setLearningStyles] = useState<string[]>([])
   219→  const [resources, setResources] = useState<string[]>([])
   220→
   221→  const now = new Date().getFullYear()
   222→  const isReview = step === STEPS.length - 1
   223→  const weeklyHours = Math.round((weekdayHours * 5 + weekendHours * 2) * 10) / 10
   224→  const progress = Math.round(((step + 1) / STEPS.length) * 100)
   225→
   226→  const stepValid = useMemo(() => {
   227→    switch (step) {
   228→      case 0:
   229→        return name.trim().length > 0
   230→      case 1:
   231→        return year != null && Number.isFinite(gradYear) && gradYear >= now && gradYear <= now + 12
   232→      case 2:
   233→        return prepStage != null
   234→      case 3:
   235→        return dailyHours >= 0.5 && dailyHours <= 12 && weekdayHours >= 0 && weekendHours >= 0
   236→      case 4:
   237→        return learningStyles.length >= 1
   238→      default:
   239→        return true
   240→    }
   241→  }, [step, name, year, gradYear, now, prepStage, dailyHours, weekdayHours, weekendHours, learningStyles])
   242→
   243→  const reviewRows = useMemo(
   244→    () => [
   245→      { k: 'Name', v: name.trim() || 'Future Dr.' },
   246→      { k: 'Year', v: year != null ? (YEAR_LABELS[year] ?? `Year ${year}`) : '—' },
   247→      { k: 'Semester', v: `Semester ${semester}` },
   248→      { k: 'College', v: collegeName.trim() || '—' },
   249→      { k: 'College type', v: collegeType },
   250→      { k: 'Expected graduation', v: String(gradYear) },
   251→      { k: 'Past performance', v: pastScore.trim() || '—' },
   252→      { k: 'Prep stage', v: prepStage ? (PREP_STAGE_LABELS[prepStage] ?? prepStage) : '—' },
   253→      { k: 'Daily target', v: `${dailyHours} h/day` },
   254→      { k: 'Weekly capacity', v: `≈ ${weeklyHours} h/week` },
   255→      { k: 'Learning styles', v: learningStyles.join(', ') || '—' },
   256→      { k: 'Resources', v: resources.join(', ') || 'None' },
   257→    ],
   258→    [name, year, semester, collegeName, collegeType, gradYear, pastScore, prepStage, dailyHours, weeklyHours, learningStyles, resources]
   259→  )
   260→
   261→  // Choosing a year auto-derives semester and expected graduation (user can override after).
   262→  const chooseYear = (y: number) => {
   263→    setYear(y)
   264→    setSemester(Math.min((y - 1) * 2 + 1, 8))
   265→    setGradYear(now + (y <= 4 ? 5 - y : 1))
   266→  }
   267→
   268→  const toggleIn = (list: string[], v: string) =>
   269→    list.includes(v) ? list.filter((x) => x !== v) : [...list, v]
   270→
   271→  const next = () => {
   272→    if (!stepValid) return
   273→    setError(null)
   274→    setDir(1)
   275→    setStep((s) => Math.min(s + 1, STEPS.length - 1))
   276→  }
   277→
   278→  const back = () => {
   279→    setError(null)
   280→    setDir(-1)
   281→    setStep((s) => Math.max(s - 1, 0))
   282→  }
   283→
   284→  const onKeyDown = (e: ReactKeyboardEvent<HTMLDivElement>) => {
   285→    if (e.key !== 'Enter' || isReview || saving || skipping) return
   286→    const el = e.target as HTMLElement
   287→    if (el.tagName === 'BUTTON' || el.tagName === 'TEXTAREA') return
   288→    if (el.closest('[role="listbox"]')) return
   289→    if (!stepValid) return
   290→    e.preventDefault()
   291→    next()
   292→  }
   293→
   294→  const enterMedos = async () => {
   295→    if (saving) return
   296→    setSaving(true)
   297→    setError(null)
   298→    try {
   299→      const res = await api.saveProfile({
   300→        name: name.trim() || 'Future Dr.',
   301→        year: year ?? 2,
   302→        semester,
   303→        collegeName: collegeName.trim(),
   304→        collegeType,
   305→        gradYear,
   306→        pastScore: pastScore.trim(),
   307→        prepStage: prepStage ?? 'serious',
   308→        dailyHours,
   309→        weekdayHours,
   310→        weekendHours,
   311→        learningStyles,
   312→        resources,
   313→        onboarded: true,
   314→      })
   315→      setProfile(res.profile)
   316→      toast({
   317→        title: `Welcome to MEDOS, ${res.profile.name}.`,
   318→        description: 'Your learning system is calibrated. Building your first dashboard…',
   319→      })
   320→      setView('home')
   321→    } catch (err) {
   322→      setError(
   323→        err instanceof Error
   324→          ? `${err.message} — check your connection and press ENTER MEDOS again.`
   325→          : 'Something went wrong while saving your profile. Please try again.'
   326→      )
   327→      setSaving(false)
   328→    }
   329→  }
   330→
   331→  const skipToDemo = async () => {
   332→    if (skipping || saving) return
   333→    setSkipping(true)
   334→    try {
   335→      await api.saveProfile({ onboarded: true })
   336→    } catch {
   337→      toast({ title: 'Could not reach the server — continuing with the demo student.' })
   338→    }
   339→    setView('home')
   340→  }
   341→
   342→  const motionCustom = reduce ? 0 : dir
   343→
   344→  return (
   345→    <div className="relative min-h-svh overflow-hidden bg-background text-foreground">
   346→      <div className="med-grid absolute inset-0" aria-hidden />
   347→      <div
   348→        className="absolute left-1/2 top-[-15%] h-[420px] w-[680px] -translate-x-1/2 rounded-full bg-cyan-500/[0.06] blur-3xl"
   349→        aria-hidden
   350→      />
   351→
   352→      {/* Persistent skip — demo student is always one tap away */}
   353→      <div className="absolute right-4 top-4 z-10 sm:right-6 sm:top-6">
   354→        <Button
   355→          variant="ghost"
   356→          size="sm"
   357→          className="h-9 rounded-full text-xs text-muted-foreground hover:text-foreground"
   358→          onClick={skipToDemo}
   359→          disabled={skipping || saving}
   360→        >
   361→          {skipping && <Loader2 className="size-3.5 animate-spin" aria-hidden />}
   362→          Use demo student →
   363→        </Button>
   364→      </div>
   365→
   366→      <div className="relative flex min-h-svh items-center justify-center px-4 py-20 sm:py-16">
   367→        <div
   368→          className="glass-strong w-full max-w-xl rounded-3xl p-5 shadow-2xl sm:p-8"
   369→          onKeyDown={onKeyDown}
   370→        >
   371→          {/* Header: wordmark + step counter */}
   372→          <div className="flex items-center justify-between gap-3">
   373→            <div className="flex items-center gap-2.5 text-base font-bold tracking-tight">
   374→              <span className="relative flex size-2">
   375→                <span className="absolute inline-flex h-full w-full animate-ping rounded-full bg-cyan-400 opacity-60" />
   376→                <span className="relative inline-flex size-2 rounded-full bg-cyan-400" />
   377→              </span>
   378→              MEDOS
   379→            </div>
   380→            <p className="text-xs font-medium tracking-wide text-muted-foreground" aria-live="polite">
   381→              {isReview ? 'Final review' : `Step ${step + 1} of 6`}
   382→            </p>
   383→          </div>
   384→
   385→          <Progress value={progress} className="mt-4 h-1.5" />
   386→
   387→          {/* Step dots */}
   388→          <div className="mt-3 flex items-center justify-center gap-1.5" aria-hidden>
   389→            {STEPS.map((s, i) => (
   390→              <span
   391→                key={s}
   392→                className={cn(
   393→                  'h-1.5 rounded-full transition-all duration-300',
   394→                  i === step ? 'w-6 bg-primary' : 'w-1.5 bg-muted-foreground/25'
   395→                )}
   396→              />
   397→            ))}
   398→          </div>
   399→
   400→          {/* Step body */}
   401→          <div className="mt-6 min-h-[380px] sm:min-h-[400px]">
   402→            <AnimatePresence mode="wait" custom={motionCustom} initial={false}>
   403→              <motion.div

   405→                custom={motionCustom}
   406→                variants={stepVariants}
   407→                initial="enter"
   408→                animate="center"
   409→                exit="exit"

   413→                  <>
   414→                    <StepHeader
   415→                      icon={Sparkles}
   416→                      title="Welcome, future doctor."
   417→                      sub="Five minutes of setup — then MEDOS starts working like a system, not a pile of PDFs."
   418→                    />
   419→                    <div className="mt-6 space-y-2">
   420→                      <Label htmlFor="ob-name">What should we call you?</Label>
   421→                      <Input
   422→                        id="ob-name"
   423→                        value={name}
   424→                        onChange={(e) => setName(e.target.value)}
   425→                        placeholder="Your name"
   426→                        autoComplete="name"
   427→                        maxLength={60}
   428→                        className="h-11"
   429→                      />
   430→                      {name.trim().length === 0 && (
   431→                        <p className="text-xs text-sev-warn">A name is required to continue.</p>
   432→                      )}
   433→                    </div>
   434→                    <ul className="mt-6 space-y-3.5 text-sm leading-relaxed text-ink-soft">
   435→                      <li className="flex gap-3">
   436→                        <Network className="mt-0.5 size-4 shrink-0 text-cyan-400" aria-hidden />
   437→                        Your entire MBBS curriculum, connected into one navigable knowledge map.
   438→                      </li>
   439→                      <li className="flex gap-3">
   440→                        <Target className="mt-0.5 size-4 shrink-0 text-cyan-400" aria-hidden />
   441→                        Daily missions picked by an engine that knows what you knew — and what you forgot.

   443→                      <li className="flex gap-3">

   445→                        A NEET-PG roadmap that starts today, not after internship panic sets in.
   446→                      </li>
   447→                    </ul>
   448→                  </>

   450→
   451→                {step === 1 && (
   452→                  <>
   453→                    <StepHeader
   454→                      icon={GraduationCap}
   455→                      title="Where are you in MBBS?"
   456→                      sub="This calibrates the map — subjects, depth and exam weight all shift with your year."
